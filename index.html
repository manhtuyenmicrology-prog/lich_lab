const SS = SpreadsheetApp.getActiveSpreadsheet(); function doGet(e) { const p = e.parameter; const out = ContentService.createTextOutput(); out.setMimeType(ContentService.MimeType.JSON); try { let r; if (p.action === 'getSessions') r = getSessions(); else if (p.action === 'addSession') r = addSession(p); else if (p.action === 'deleteSession') r = deleteSession(p.id); else if (p.action === 'updateSession') r = updateSession(p); else if (p.action === 'join') r = joinSession(p); else if (p.action === 'leave') r = leaveSession(p); else if (p.action === 'getMembers') r = getMembers(); else if (p.action === 'getTeamMembers') r = getTeamMembers(); else if (p.action === 'getFund') r = getFund(); else if (p.action === 'addTransaction') r = addTransaction(p); else if (p.action === 'deleteTransaction') r = deleteTransaction(p.id); else r = { error: 'Unknown action' }; out.setContent(JSON.stringify(r)); } catch(err) { out.setContent(JSON.stringify({ error: err.message })); } return out; } function doPost(e) { return doGet(e); } function getSheet(name) { let s = SS.getSheetByName(name); if (!s) { s = SS.insertSheet(name); if (name === 'sessions') s.appendRow(['id','group','date','slot','title','content','created_at']); if (name === 'participants') s.appendRow(['id','session_id','name','joined_at']); if (name === 'members') s.appendRow(['group','name','role','year','research','contact','avatar_url']); if (name === 'team_members') s.appendRow(['group','name','role','year']); if (name === 'fund') s.appendRow(['id','type','amount','desc','date','created_at']); } return s; } // Fix timezone: dùng timezone Việt Nam (UTC+7) function fmtDate(val) { if (val instanceof Date) { // Chuyển sang giờ VN const vn = new Date(val.getTime() + 7 * 60 * 60 * 1000); const y = vn.getUTCFullYear(); const m = String(vn.getUTCMonth()+1).padStart(2,'0'); const d = String(vn.getUTCDate()).padStart(2,'0'); return y + '-' + m + '-' + d; } if (typeof val === 'string') return val.slice(0,10); return String(val).slice(0,10); } function getSessions() { const sSheet = getSheet('sessions'); const pSheet = getSheet('participants'); const sData = sSheet.getDataRange().getValues().slice(1).filter(r => r[0] !== ''); const pData = pSheet.getDataRange().getValues().slice(1).filter(r => r[0] !== ''); return sData.map(r => ({ id: r[0], group: r[1], date: fmtDate(r[2]), slot: r[3], title: r[4], content: r[5], created_at: r[6], participants: pData.filter(p => p[1] == r[0]).map(p => ({ id: p[0], name: p[2] })) })); } function addSession(p) { const sheet = getSheet('sessions'); const id = Date.now().toString(); sheet.appendRow([id, p.group, p.date, p.slot, p.title, p.content || '', new Date().toISOString()]); return { success: true, id }; } function deleteSession(id) { const sSheet = getSheet('sessions'); const pSheet = getSheet('participants'); const sData = sSheet.getDataRange().getValues(); for (let i = 1; i < sData.length; i++) { if (sData[i][0] == id) { sSheet.deleteRow(i + 1); break; } } const pData = pSheet.getDataRange().getValues(); for (let i = pData.length - 1; i >= 1; i--) { if (pData[i][1] == id) pSheet.deleteRow(i + 1); } return { success: true }; } function updateSession(p) { const sheet = getSheet('sessions'); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] == p.id) { if (p.title) sheet.getRange(i+1, 5).setValue(p.title); if (p.content !== undefined) sheet.getRange(i+1, 6).setValue(p.content); if (p.date) sheet.getRange(i+1, 3).setValue(p.date); if (p.slot !== undefined) sheet.getRange(i+1, 4).setValue(p.slot); if (p.group) sheet.getRange(i+1, 2).setValue(p.group); return { success: true }; } } return { error: 'Not found' }; } function joinSession(p) { const sheet = getSheet('participants'); const data = sheet.getDataRange().getValues().slice(1); if (data.find(r => r[1] == p.session_id && r[2] === p.name)) return { error: 'Bạn đã tham gia ca này rồi!' }; const id = Date.now().toString(); sheet.appendRow([id, p.session_id, p.name, new Date().toISOString()]); return { success: true, id }; } function leaveSession(p) { const sheet = getSheet('participants'); const data = sheet.getDataRange().getValues(); for (let i = data.length - 1; i >= 1; i--) { if (data[i][0] == p.id) { sheet.deleteRow(i + 1); return { success: true }; } } return { error: 'Not found' }; } function getMembers() { const sheet = getSheet('members'); const data = sheet.getDataRange().getValues().slice(1).filter(r => r[0] !== ''); return data.map(r => ({ group: r[0], name: r[1], role: r[2], year: r[3], research: r[4], contact: r[5], avatar_url: r[6] })); } function getTeamMembers() { const sheet = getSheet('team_members'); const data = sheet.getDataRange().getValues().slice(1).filter(r => r[0] !== ''); return data.map(r => ({ group: r[0], name: r[1], role: r[2], year: r[3] })); } function getFund() { const sheet = getSheet('fund'); const data = sheet.getDataRange().getValues().slice(1).filter(r => r[0] !== ''); return data.map(r => ({ id: r[0], type: r[1], amount: r[2], desc: r[3], date: fmtDate(r[4]), created_at: r[5] })); } function addTransaction(p) { const sheet = getSheet('fund'); const id = Date.now().toString(); sheet.appendRow([id, p.type, +p.amount, p.desc, p.date, new Date().toISOString()]); return { success: true, id }; } function deleteTransaction(id) { const sheet = getSheet('fund'); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] == id) { sheet.deleteRow(i + 1); return { success: true }; } } return { error: 'Not found' }; }
