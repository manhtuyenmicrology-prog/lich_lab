const SS = SpreadsheetApp.getActiveSpreadsheet(); function doGet(e) { const p = e.parameter; const out = ContentService.createTextOutput(); out.setMimeType(ContentService.MimeType.JSON); try { let r; if (p.action === 'getSessions') r = getSessions(); else if (p.action === 'addSession') r = addSession(p); else if (p.action === 'deleteSession') r = deleteSession(p.id); else if (p.action === 'join') r = joinSession(p); else if (p.action === 'leave') r = leaveSession(p); else r = { error: 'Unknown action' }; out.setContent(JSON.stringify(r)); } catch(err) { out.setContent(JSON.stringify({ error: err.message })); } return out; } function doPost(e) { return doGet(e); } function getSheet(name) { let s = SS.getSheetByName(name); if (!s) { s = SS.insertSheet(name); if (name === 'sessions') s.appendRow(['id','group','date','slot','title','content','created_at']); if (name === 'participants') s.appendRow(['id','session_id','name','joined_at']); } return s; } function getSessions() { const sSheet = getSheet('sessions'); const pSheet = getSheet('participants'); const sData = sSheet.getDataRange().getValues().slice(1).filter(r => r[0] !== ''); const pData = pSheet.getDataRange().getValues().slice(1).filter(r => r[0] !== ''); return sData.map(r => ({ id: r[0], group: r[1], date: r[2], slot: r[3], title: r[4], content: r[5], created_at: r[6], participants: pData.filter(p => p[1] == r[0]).map(p => ({ id: p[0], name: p[2] })) })); } function addSession(p) { const sheet = getSheet('sessions'); const id = Date.now().toString(); sheet.appendRow([id, p.group, p.date, p.slot, p.title, p.content, new Date().toISOString()]); return { success: true, id }; } function deleteSession(id) { const sSheet = getSheet('sessions'); const pSheet = getSheet('participants'); const sData = sSheet.getDataRange().getValues(); for (let i = 1; i < sData.length; i++) { if (sData[i][0] == id) { sSheet.deleteRow(i + 1); break; } } const pData = pSheet.getDataRange().getValues(); for (let i = pData.length - 1; i >= 1; i--) { if (pData[i][1] == id) pSheet.deleteRow(i + 1); } return { success: true }; } function joinSession(p) { const sheet = getSheet('participants'); const data = sheet.getDataRange().getValues().slice(1); if (data.find(r => r[1] == p.session_id && r[2] === p.name)) return { error: 'Bạn đã tham gia ca này rồi!' }; const id = Date.now().toString(); sheet.appendRow([id, p.session_id, p.name, new Date().toISOString()]); return { success: true, id }; } function leaveSession(p) { const sheet = getSheet('participants'); const data = sheet.getDataRange().getValues(); for (let i = data.length - 1; i >= 1; i--) { if (data[i][0] == p.id) { sheet.deleteRow(i + 1); return { success: true }; } } return { error: 'Not found' }; }
